PROGRAM 01

Selection Sort

C++ SOURCE CODE
#include <iostream>
using namespace std;

class SelectionSort {
    int a[100], n, i, j, pos, temp;

public:
    void getData() {
        cout << "Enter the number of elements: ";
        cin >> n;
        cout << "Enter " << n << " elements:" << endl;
        for (i = 0; i < n; i++) {
            cin >> a[i];
        }
    }

    void sort() {
        for (i = 0; i < n - 1; i++) {
            pos = i;
            for (j = i + 1; j < n; j++) {
                if (a[j] < a[pos]) {
                    pos = j;
                }
            }
            temp = a[i];
            a[i] = a[pos];
            a[pos] = temp;
        }
    }

    void display() {
        cout << "The sorted elements using the Selection Sort technique are:" << endl;
        for (i = 0; i < n; i++) {
            cout << a[i] << " ";
        }
        cout << endl;
    }
};

int main() {
    SelectionSort S;
    S.getData();
    S.sort();
    S.display();
    return 0;
}
PROGRAM 02

Bubble Sort

C++ SOURCE CODE
#include <iostream>
using namespace std;

class BubbleSort {
    int a[100], n, i, j, temp;

public:
    void getData() {
        cout << "Enter the number of elements: ";
        cin >> n;
        cout << "Enter " << n << " elements:" << endl;
        for (i = 0; i < n; i++) {
            cin >> a[i];
        }
    }

    void sort() {
        for (i = 0; i < n - 1; i++) {
            for (j = 0; j < n - i - 1; j++) {
                if (a[j] > a[j + 1]) {
                    temp = a[j];
                    a[j] = a[j + 1];
                    a[j + 1] = temp;
                }
            }
        }
    }

    void display() {
        cout << "The sorted elements using the Bubble Sort technique are:" << endl;
        for (i = 0; i < n; i++) {
            cout << a[i] << " ";
        }
        cout << endl;
    }
};

int main() {
    BubbleSort B;
    B.getData();
    B.sort();
    B.display();
    return 0;
}
PROGRAM 03

Linear Search

C++ SOURCE CODE
#include <iostream>
using namespace std;

int main() {
    int a[20], n, key, pos = -1, i;

    cout << "Enter the number of elements: ";
    cin >> n;
    cout << "Enter " << n << " elements:" << endl;
    for (i = 0; i < n; i++) {
        cin >> a[i];
    }

    cout << "Enter the element to be searched: ";
    cin >> key;

    for (i = 0; i < n; i++) {
        if (a[i] == key) {
            pos = i;
            break;
        }
    }

    if (pos != -1)
        cout << "Element found at position " << pos + 1 << "." << endl;
    else
        cout << "Element not found in the list." << endl;

    return 0;
}
PROGRAM 04

Towers of Hanoi

C++ SOURCE CODE
#include <iostream>
using namespace std;

void TOH(int n, char source, char auxiliary, char destination) {
    if (n == 1) {
        cout << "Move Disk 1 from " << source
             << " to " << destination << endl;
    } else {
        TOH(n - 1, source, destination, auxiliary);
        cout << "Move Disk " << n << " from " << source
             << " to " << destination << endl;
        TOH(n - 1, auxiliary, source, destination);
    }
}

int main() {
    int n;
    cout << "Enter the number of disks: ";
    cin >> n;
    cout << endl;
    cout << "Steps to solve Towers of Hanoi:" << endl;
    TOH(n, 'A', 'B', 'C');
    return 0;
}
PROGRAM 05

Dynamic Array – Smallest and Largest Element

C++ SOURCE CODE
#include <iostream>
using namespace std;

int main() {
    int n, i, *arr, max, min;

    cout << "Enter the size of the array: ";
    cin >> n;

    arr = new int[n];

    cout << "Enter the array elements:" << endl;
    for (i = 0; i < n; i++) {
        cin >> arr[i];
    }

    min = arr[0];
    max = arr[0];

    for (i = 0; i < n; i++) {
        if (arr[i] < min) {
            min = arr[i];
        }
        if (arr[i] > max) {
            max = arr[i];
        }
    }

    cout << "The maximum element in the array is: " << max << endl;
    cout << "The minimum element in the array is: " << min << endl;

    delete[] arr;
    return 0;
}
